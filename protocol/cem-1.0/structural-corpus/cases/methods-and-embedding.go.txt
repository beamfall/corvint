package cases

type base struct{ n int }

func (b *base) inc() { b.n++ }

type derived struct {
	base
	name string
}

func (d derived) String() string { return d.name }

func newDerived() *derived {
	d := &derived{name: "x"}
	d.inc()
	return d
}

func useDerived() string { return newDerived().String() }
