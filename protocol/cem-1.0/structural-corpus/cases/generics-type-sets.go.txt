package cases

import (
	"fmt"
	"cmp"
)

type Number interface {
	~int | ~int64 | ~float64
}

func sum[T Number](xs ...T) T {
	var total T
	for _, x := range xs {
		total += x
	}
	return total
}

func maxOf[T cmp.Ordered](a, b T) T {
	if a > b {
		return a
	}
	return b
}

func Show() { fmt.Println(sum(1, 2, 3), maxOf("a", "b")) }
